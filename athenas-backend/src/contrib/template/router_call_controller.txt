log = getLogger(__name__)

ctr = {{ controller }}(request, response, '{{ response_format }}')
if isinstance(ctr, controller.DefaultController):

    type = 'TEXT'

    if getattr(ctr.{{ action }}, '__update_timeout_session__', True):

        log.info('{{ controller }}.{{ action }} not update timeout session.')

    if getattr(ctr.{{ action }}, '_is_public', False) or request.user.is_active:
        t_start = x_time()
        ctr.{{ action }}({{ args | safe }})
        t_end = x_time()
        ctr.response['Response-Time'] = '%0.3fms' % ((t_end - t_start) * 1000.0)
    else:
        ctr = Application(request, response)
        ctr.un_autorized((type,))
